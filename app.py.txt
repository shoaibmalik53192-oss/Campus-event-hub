import streamlit as st
from pypdf import PdfReader
from youtube_transcript_api import YouTubeTranscriptApi
from google import genai

st.set_page_config(page_title="AI Learning Assistant", layout="wide")
st.title("📚 AI Personal Learning Assistant")

# Session state initialize karein taaki data tab switch karne par lost na ho
if "summary" not in st.session_state:
    st.session_state.summary = None
if "quiz" not in st.session_state:
    st.session_state.quiz = None

api_key = st.sidebar.text_input("Gemini API Key Enter Karein:", type="password")

if api_key:
    client = genai.Client(api_key=api_key)
    option = st.radio("Content Source Select Karein:", ["PDF File", "YouTube Link"])
    text_content = ""

    if option == "PDF File":
        uploaded_file = st.file_uploader("PDF Upload Karein", type=["pdf"])
        if uploaded_file:
            reader = PdfReader(uploaded_file)
            for page in reader.pages:
                text_content += page.extract_text() or ""
            st.success("PDF Read Ho Gaya!")

    elif option == "YouTube Link":
        youtube_url = st.text_input("YouTube Video URL Daalein:")
        if youtube_url and "v=" in youtube_url:
            video_id = youtube_url.split("v=")[1].split("&")[0]
            try:
                transcript = YouTubeTranscriptApi.get_transcript(video_id)
                text_content = " ".join([t['text'] for t in transcript])
                st.success("Transcript Fetch Ho Gaya!")
            except Exception as e:
                st.error("Transcript mil nahi paaya. Subtitles check karein.")

    # Button click par results generate karein
    if text_content and st.button("Generate Study Notes"):
        with st.spinner("AI Material Generate Kar Raha Hai..."):
            try:
                summary_prompt = f"Is text ki top 5 bullet point summary banao:\n\n{text_content[:4000]}"
                summary_res = client.models.generate_content(
                    model='gemini-2.5-flash',
                    contents=summary_prompt
                )

                quiz_prompt = f"Is text se 3 practice multiple choice questions (MCQ) banao correct answer ke saath:\n\n{text_content[:4000]}"
                quiz_res = client.models.generate_content(
                    model='gemini-2.5-flash',
                    contents=quiz_prompt
                )

                # Results ko session state me save karein
                st.session_state.summary = summary_res.text
                st.session_state.quiz = quiz_res.text
            except Exception as e:
                st.error(f"Error: {e}")

    # Tabs ko button ke bahar render karein taaki results screen par bane rahein
    if st.session_state.summary or st.session_state.quiz:
        tab1, tab2 = st.tabs(["📝 Key Summary", "❓ Practice Quiz"])
        
        with tab1:
            st.subheader("Summary Points")
            st.write(st.session_state.summary)

        with tab2:
            st.subheader("Quiz Questions")
            st.write(st.session_state.quiz)

else:
    st.warning("Sidebar me apni Gemini API Key daal kar start karein!")
